function setupSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  resetLog();

  // --- Jobs tab ---
  let jobs = ss.getSheetByName(SHEET_JOBS);
  if (!jobs) jobs = ss.insertSheet(SHEET_JOBS);
  if (jobs.getLastRow() === 0) {
    jobs.appendRow(['JobName', 'Active', 'Address', 'Notes']);
    jobs.appendRow(['Sample Job 1 — 12 Smith St', 'Y', '', '']);
    jobs.appendRow(['Sample Job 2 — 4 Jones Rd', 'Y', '', '']);
    jobs.setFrozenRows(1);
  }

  // --- Touchups tab ---
  let touch = ss.getSheetByName(SHEET_TOUCHUPS);
  if (!touch) touch = ss.insertSheet(SHEET_TOUCHUPS);
  if (touch.getLastRow() === 0) {
    touch.appendRow([
      'Timestamp', 'Job', 'Type', 'Item', 'Description',
      'Status', 'AddedBy', 'SupervisorSignoff', 'DateCompleted'
    ]);
    touch.setFrozenRows(1);
  }

  // --- Weekly KPI tab ---
  addWeeklyKpiTab();

  // --- JSA tab ---
  addJsaTab();

  // --- Roster tab (SMS reminders) ---
  addRosterTab();

  SpreadsheetApp.flush();
  Logger.log('Sheets ready: Log, Jobs, Touchups, WeeklyKPI, JSA, Roster');
}

function installTriggers() {
  // Remove any existing processEodEmails triggers first (avoid duplicates)
  ScriptApp.getProjectTriggers().forEach(t => {
    if (t.getHandlerFunction() === 'processEodEmails') {
      ScriptApp.deleteTrigger(t);
    }
  });
  ScriptApp.newTrigger('processEodEmails')
    .timeBased()
    .everyMinutes(10)
    .create();
  Logger.log('Trigger installed: processEodEmails every 10 minutes');
}

// Run this once after you've set the Twilio Script Properties (see
// sendSms below) and populated the Roster tab. Installs the reminder
// schedule (all times are in the project's timezone — set to
// Australia/Hobart in Project Settings so these land at the right
// local time in Tasmania):
//   - 7:00am Mon-Fri  -> everyone   -> Morning JSA sign-on
//   - 2:55pm Mon-Fri  -> everyone   -> EOD checklist
//   - 4:00pm Friday   -> supervisors only -> Weekly KPI scorecard
function installReminderTriggers() {
  ['sendMorningReminder', 'sendAfternoonReminder', 'sendFridayKpiReminder'].forEach(fn => {
    ScriptApp.getProjectTriggers().forEach(t => {
      if (t.getHandlerFunction() === fn) ScriptApp.deleteTrigger(t);
    });
  });
  ScriptApp.newTrigger('sendMorningReminder')
    .timeBased().atHour(7).nearMinute(0).everyDays(1).create();
  ScriptApp.newTrigger('sendAfternoonReminder')
    .timeBased().atHour(14).nearMinute(55).everyDays(1).create();
  ScriptApp.newTrigger('sendFridayKpiReminder')
    .timeBased().onWeekDay(ScriptApp.WeekDay.FRIDAY).atHour(16).nearMinute(0).create();
  Logger.log('Reminder triggers installed: 7:00am + 2:55pm Mon-Fri, 4:00pm Friday (supervisors only)');
}

function doGet(e) {
  const action = e.parameter.action;
  let result;
  try {
    if (action === 'jobs') {
      result = getActiveJobs();
    } else if (action === 'log') {
      result = getLogForDate(e.parameter.date);
    } else if (action === 'punchlist') {
      result = getPunchlistForJob(e.parameter.job);
    } else if (action === 'summary') {
      result = getSupervisorSummary(e.parameter.date);
    } else {
      result = { error: 'Unknown action: ' + action };
    }
  } catch (err) {
    result = { error: err.message };
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  let result;
  try {
    const data = JSON.parse(e.postData.contents);
    if (data.action === 'eod') {
      result = writeEodRow(data);
    } else if (data.action === 'punchlist_add') {
      result = addPunchlistItem(data);
    } else if (data.action === 'punchlist_update') {
      result = updatePunchlistItem(data);
    } else if (data.action === 'stage_signoff') {
      result = addStageSignoff(data);
    } else if (data.action === 'weekly_kpi') {
      result = writeWeeklyKpiRow(data);
    } else if (data.action === 'jsa_signon') {
      result = writeJsaSignon(data);
    } else {
      result = { error: 'Unknown action: ' + data.action };
    }
  } catch (err) {
    result = { error: err.message };
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function getActiveJobs() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_JOBS);
  const rows = sheet.getDataRange().getValues();
  const jobs = [];
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][1]).toUpperCase() === 'Y') {
      jobs.push(rows[i][0]);
    }
  }
  return { jobs };
}

function getLogForDate(dateStr) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_LOG);
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0];
  const targetDate = dateStr || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  const entries = [];
  for (let i = 1; i < rows.length; i++) {
    const rowDate = rows[i][1]; // 'Date' column
    if (String(rowDate) === targetDate) {
      const obj = {};
      headers.forEach((h, idx) => obj[h] = rows[i][idx]);
      entries.push(obj);
    }
  }
  return { date: targetDate, entries };
}

function getSupervisorSummary(dateStr) {
  const logData = getLogForDate(dateStr);
  const submittedNames = logData.entries.flatMap(e => String(e.Name || '').split(',').map(s => s.trim()).filter(Boolean));
  let totalVariationHours = 0;
  const flags = [];
  logData.entries.forEach(e => {
    if (e.VariationHours) totalVariationHours += Number(e.VariationHours) || 0;
    if (e.FlagDamage) flags.push({ type: 'Damage', name: e.Name, job: e.Job, detail: e.FlagDamageDetail });
    if (e.FlagVariation) flags.push({ type: 'Variation', name: e.Name, job: e.Job, detail: e.FlagVariationDetail });
    if (e.FlagHoldUp) flags.push({ type: 'Hold-up', name: e.Name, job: e.Job, detail: e.FlagHoldUpDetail });
    if (e.FlagMaterials) flags.push({ type: 'Materials', name: e.Name, job: e.Job, detail: e.FlagMaterialsDetail });
    if (e.SafetyIncident === 'Yes') flags.push({ type: 'SAFETY', name: e.Name, job: e.Job, detail: e.SafetyIncidentDetail });
    if (e.QualityIssue === 'Yes') flags.push({ type: 'Quality/defect', name: e.Name, job: e.Job, detail: e.QualityIssueDetail });
    if (e.ScheduleImpact === 'Yes') flags.push({ type: 'Schedule impact', name: e.Name, job: e.Job, detail: e.ScheduleImpactDetail });
    if (e.OfficeAction === 'Yes') flags.push({ type: 'Office action needed', name: e.Name, job: e.Job, detail: e.OfficeActionDetail });
    if (e.MaterialsToOrder === 'Yes') flags.push({ type: 'Materials to order', name: e.Name, job: e.Job, detail: e.MaterialsToOrderDetail });
  });
  return {
    date: logData.date,
    submitted: submittedNames,
    totalVariationHours,
    flags,
    entryCount: logData.entries.length
  };
}

function getPunchlistForJob(jobName) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_TOUCHUPS);
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0];
  const items = [];
  for (let i = 1; i < rows.length; i++) {
    if (rows[i][1] === jobName) {
      const obj = {};
      headers.forEach((h, idx) => obj[h] = rows[i][idx]);
      items.push(obj);
    }
  }
  return { job: jobName, items };
}

function updatePunchlistItem(d) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_TOUCHUPS);
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === d.timestamp && rows[i][1] === d.job && rows[i][3] === d.item) {
      sheet.getRange(i + 1, 6).setValue(d.status || rows[i][5]); // Status col
      if (d.status === 'Done') {
        sheet.getRange(i + 1, 9).setValue(new Date()); // DateCompleted col
      }
      return { ok: true };
    }
  }
  return { error: 'Item not found' };
}

// Credentials are NOT stored in this file. Set them once via the Apps
// Script editor: Project Settings (gear icon) -> Script Properties ->
// add TWILIO_SID, TWILIO_AUTH_TOKEN, TWILIO_FROM_NUMBER (your Twilio
// SMS-capable number, e.g. +1XXXXXXXXXX).

// Add crew names/numbers to the Roster tab (Active = Y). Then run
// installReminderTriggers() once from the editor.

function isWeekday_() {
  const day = new Date().getDay(); // 0 = Sun, 6 = Sat
  return day >= 1 && day <= 5;
}

// 7:00am Mon-Fri, everyone — link straight to the Morning JSA tab
function sendMorningReminder() {
  if (!isWeekday_()) return;
  sendSms(`Morning! Please complete your Morning JSA sign-on before starting work: ${APP_URL}#morning`);
}

// 2:55pm Mon-Fri, everyone — link straight to the Crew (EOD) tab
function sendAfternoonReminder() {
  if (!isWeekday_()) return;
  sendSms(`End of day reminder: please complete your EOD checklist before knocking off: ${APP_URL}#crew`);
}

// 4:00pm Friday only, supervisors only — link to the Supervisor (KPI) tab.
// The trigger itself only fires on Fridays, but the day check is a
// harmless safety net in case that ever changes.
function sendFridayKpiReminder() {
  if (new Date().getDay() !== 5) return;
  sendSms(`Friday reminder: please complete the Weekly Crew Leader KPI Scorecard: ${APP_URL}#sup`, 'Supervisor');
}

// roleFilter (optional): 'Supervisor' or 'Carpenter' — only texts Roster
// rows whose Role column matches (case-insensitive). Omit to text everyone.
function sendSms(message, roleFilter) {
  const props = PropertiesService.getScriptProperties();
  const sid = props.getProperty('TWILIO_SID');
  const token = props.getProperty('TWILIO_AUTH_TOKEN');
  const from = props.getProperty('TWILIO_FROM_NUMBER');
  if (!sid || !token || !from) {
    Logger.log('Twilio credentials not set in Script Properties — skipping SMS send.');
    return;
  }

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_ROSTER);
  if (!sheet) { Logger.log('Roster tab not found — run addRosterTab() first.'); return; }

  const rows = sheet.getDataRange().getValues();
  const url = `https://api.twilio.com/2010-04-01/Accounts/${sid}/Messages.json`;
  const authHeader = 'Basic ' + Utilities.base64Encode(sid + ':' + token);

  let sent = 0;
  for (let i = 1; i < rows.length; i++) {
    const [name, phone, role, active] = rows[i];
    if (String(active).toUpperCase() !== 'Y' || !phone) continue;
    if (roleFilter && String(role).trim().toLowerCase() !== roleFilter.toLowerCase()) continue;
    try {
      const resp = UrlFetchApp.fetch(url, {
        method: 'post',
        payload: { To: String(phone), From: from, Body: message },
        headers: { Authorization: authHeader },
        muteHttpExceptions: true
      });
      if (resp.getResponseCode() >= 300) {
        Logger.log(`SMS failed for ${name} (${phone}): ${resp.getContentText()}`);
      } else {
        sent++;
      }
    } catch (err) {
      Logger.log(`SMS failed for ${name} (${phone}): ${err.message}`);
    }
  }
  Logger.log(`SMS reminder sent to ${sent} of ${rows.length - 1} roster entries.`);
}

// EMAIL — reliability pattern: trigger scans for unemailed rows
// ============================================================
function processEodEmails() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_LOG);
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0];
  const emailedCol = headers.indexOf('Emailed') + 1; // 1-based

  for (let i = 1; i < rows.length; i++) {
    if (rows[i][emailedCol - 1]) continue; // already emailed
    const obj = {};
    headers.forEach((h, idx) => obj[h] = rows[i][idx]);
    try {
      const flagBits = [];
      if (obj.SafetyIncident === 'Yes') flagBits.push('SAFETY');
      if (obj.QualityIssue === 'Yes') flagBits.push('DEFECT');
      const flagPrefix = flagBits.length ? '[' + flagBits.join('/') + '] ' : '';
      const subject = `${flagPrefix}${BUSINESS_NAME} EOD — ${obj.Name} — ${obj.Job} (${obj.Date})`;
      MailApp.sendEmail({
        to: EMAIL_TO.join(','),
        subject,
        htmlBody: formatEod(obj)
      });
      sheet.getRange(i + 1, emailedCol).setValue(new Date());
    } catch (err) {
      Logger.log('Email failed for row ' + (i + 1) + ': ' + err.message);
      // leave Emailed blank — will retry next run
    }
  }
}

function formatEod(o) {
  let html = `<h2>${o.Name} (${o.Role}) — ${o.Job}</h2>`;

  html += `<div style="background:#fff8dc;border:1px solid #e0c200;padding:10px;margin:10px 0;">
    <b>Job update:</b><br>${(o.JobUpdate || '—').replace(/\n/g, '<br>')}
  </div>`;

  if (o.SafetyIncident === 'Yes') {
    html += `<div style="background:#fdecea;border:2px solid #c0392b;padding:10px;margin:10px 0;">
      <b>⚠️ SAFETY INCIDENT / HAZARD / NEAR MISS:</b><br>${o.SafetyIncidentDetail || '—'}
    </div>`;
  }

  const flags = [];
  if (o.FlagDamage) flags.push(`<li><b>Damage / make-good:</b> ${o.FlagDamageDetail || '—'}</li>`);
  if (o.FlagVariation) flags.push(`<li><b>Variation</b> (${o.VariationHours || 0} hrs): ${o.FlagVariationDetail || '—'}</li>`);
  if (o.FlagHoldUp) flags.push(`<li><b>Hold-up:</b> ${o.FlagHoldUpDetail || '—'}</li>`);
  if (o.FlagMaterials) flags.push(`<li><b>Materials low:</b> ${o.FlagMaterialsDetail || '—'}</li>`);
  if (flags.length) {
    html += `<div style="background:#fdecea;border:1px solid #c0392b;border-radius:6px;padding:8px 12px;margin:10px 0;"><p style="margin:0 0 4px;"><b>🚩 Flags raised:</b></p><ul style="margin:0;">${flags.join('')}</ul></div>`;
  } else {
    html += `<p><b style="color:#2e7d32;">✅ Flags raised:</b> none</p>`;
  }

  html += `<p><b>Daily checklist:</b></p><ul>`;
  html += formatCheckLine('Scheduled Tradify tasks completed', o.TasksCompleted, o.TasksOutstanding, 'No');
  html += formatCheckLine('Anything affecting schedule/timeline', o.ScheduleImpact, o.ScheduleImpactDetail, 'Yes');
  html += formatCheckLine('Photos taken', o.PhotosTaken, '', 'No');
  html += formatCheckLine('Progress photos uploaded to Tradify', o.PhotosUploaded, '', 'No');
  html += formatCheckLine('Tradify timesheets complete & accurate', o.TimesheetsComplete, o.TimesheetsNote, 'No');
  html += formatCheckLine('Client happy with progress', o.ClientHappy, o.ClientHappyDetail, 'No', 'N/A');
  html += formatCheckLine('Office needs to action before tomorrow', o.OfficeAction, o.OfficeActionDetail, 'Yes');
  html += formatCheckLine('Materials need to be ordered', o.MaterialsToOrder, o.MaterialsToOrderDetail, 'Yes');
  html += formatCheckLine('Ready for tomorrow\'s work', o.ReadyTomorrow, o.ReadyTomorrowDetail, 'No');
  html += formatCheckLine('Defects / damage / quality issues found', o.QualityIssue, o.QualityIssueDetail, 'Yes');
  html += formatCheckLine('Job site clean & tidy', o.SiteClean, o.SiteCleanNote, 'No', 'N/A');
  html += formatCheckLine('Tools packed', o.ToolsPacked, o.ToolsPackedNote, 'No', 'N/A');
  html += formatCheckLine('Plant / equipment secured', o.PlantSecured, o.PlantSecuredNote, 'No', 'N/A');
  html += formatCheckLine('Rubbish removed / stored', o.WasteSorted, o.WasteSortedNote, 'No', 'N/A');
  html += formatCheckLine('Good day', o.GoodDay, o.GoodDayDetail, 'No');
  html += `</ul>`;

  if (o.OtherNotes) html += `<p><b>Other notes:</b> ${o.OtherNotes}</p>`;
  if (o.ApprenticeReflection) html += `<p><b>Apprentice reflection:</b> ${o.ApprenticeReflection}</p>`;

  return html;
}

// Generic line formatter for the daily checklist.
// alertValue / alertValue2: the answer value(s) that should show the note (if present).
function formatCheckLine(label, value, note, alertValue, alertValue2) {
  const isAlert = value === alertValue || (alertValue2 && value === alertValue2);
  const color = isAlert ? '#c0392b' : '#2e7d32';
  const dot = isAlert ? '🔴' : '🟢';
  let line = `<li>${dot} ${label}: <b style="color:${color}">${value || '—'}</b>`;
  if (note && isAlert) {
    line += ` — ${note}`;
  }
  line += `</li>`;
  return line;
}

function formatWeeklyKpi(d) {
  let html = `<h2>Weekly KPI Scorecard — ${d.crewLeader}</h2>`;
  html += `<p><b>Week:</b> ${d.weekStart} to ${d.weekEnd}</p>`;

  html += `<div style="background:#fff8dc;border:1px solid #e0c200;padding:10px;margin:10px 0;">
    <b>Total score:</b> ${d.totalScore} / 45<br>
    <b>Rating:</b> ${d.performanceRating}<br>
    <b>Bonus earned:</b> $${d.bonusEarned}
  </div>`;

  const kpiLines = [
    ['Safety', d.safety, d.safetyNote],
    ['Productivity', d.productivity, d.productivityNote],
    ['Quality', d.quality, d.qualityNote],
    ['Leadership', d.leadership, d.leadershipNote],
    ['Communication', d.communication, d.communicationNote],
    ['Company standards', d.companyStandards, d.companyStandardsNote],
    ['Ownership & initiative', d.ownershipInitiative, d.ownershipInitiativeNote],
    ['Call backs for defects', d.callBacksDefects, d.callBacksDefectsNote],
    ['Un-necessary calls', d.unnecessaryCalls, d.unnecessaryCallsNote]
  ];
  html += `<p><b>KPI ratings (out of 5):</b></p><ul>`;
  kpiLines.forEach(([label, value, note]) => {
    html += `<li>${label}: <b>${value || '—'}</b>${note ? ' — ' + note : ''}</li>`;
  });
  html += `</ul>`;

  html += `<p><b>Everyone's hours done and correct:</b> ${d.hoursCorrect || '—'}</p>`;
  html += `<p><b>Receipts uploaded correctly with job number (JN):</b> ${d.receiptsUploaded || '—'}</p>`;

  if (d.biggestWin) html += `<p><b>Biggest win this week:</b> ${d.biggestWin}</p>`;
  if (d.biggestChallenge) html += `<p><b>Biggest challenge this week:</b> ${d.biggestChallenge}</p>`;
  if (d.areasImprovement) html += `<p><b>Areas for improvement:</b> ${d.areasImprovement}</p>`;
  if (d.goalsNextWeek) html += `<p><b>Goals for next week:</b> ${d.goalsNextWeek}</p>`;
  if (d.officeSupport) html += `<p><b>How the office can help:</b> ${d.officeSupport}</p>`;

  return html;
}
